assert(not game:GetService("RunService"):IsRunning(), "Primero pulsa Stop.")
local RS = game:GetService("ReplicatedStorage")
local SSS = game:GetService("ServerScriptService")
local SPS = game:GetService("StarterPlayer").StarterPlayerScripts
local storage = game:GetService("ServerStorage")
local editor = game:GetService("ScriptEditorService")
local folder = RS:FindFirstChild("HR_Cine")
assert(folder and folder:GetAttribute("HR_Owner") == "HR_Cine_V2", "No se reconoce HR_Cine.")
local package
for _, parent in ipairs({workspace, storage}) do
 for _, candidate in ipairs(parent:GetChildren()) do
  if candidate.Name == "HR_EntregaV6" then
   local id = candidate:FindFirstChild("Identificador")
   local server = candidate:FindFirstChild("HR_CineServidor")
   local client = candidate:FindFirstChild("HR_CineCamara")
   if id and id:IsA("StringValue") and id.Value == "HR_ENTREGA_V6_20261103"
    and server and server:IsA("Script") and client and client:IsA("LocalScript")
    and server.Source:find("candidateDirections", 1, true)
    and client.Source:find('remote:FireServer("READY")', 1, true) then
    package = candidate
    break
   end
  end
 end
 if package then break end
end
assert(package, "No se encuentra el paquete corregido. Importalo primero.")
local entries = {
 {folder, "Configuracion", "ModuleScript"},
 {SSS, "HR_CineServidor", "Script"},
 {SPS, "HR_CineCamara", "LocalScript"},
}
for _, entry in ipairs(entries) do
 entry.target = entry[1]:FindFirstChild(entry[2])
 local source = package:FindFirstChild(entry[2])
 assert(entry.target and entry.target:IsA(entry[3])
  and entry.target:GetAttribute("HR_Owner") == "HR_Cine_V2", "Destino no reconocido: " .. entry[2])
 assert(source and source:IsA(entry[3]), "Paquete incompleto: " .. entry[2])
 entry.text = source.Source
end
local backup = Instance.new("Folder")
backup.Name = "HR_Respaldo_Editor_" .. tostring(os.time())
for _, entry in ipairs(entries) do
 local text = Instance.new("StringValue")
 text.Name = entry[2]
 text.Value = editor:GetEditorSource(entry.target)
 text.Parent = backup
end
backup.Parent = storage
for _, entry in ipairs(entries) do
 local sourceText = entry.text
 editor:UpdateSourceAsync(entry.target, function() return sourceText end)
end
for _, entry in ipairs(entries) do
 assert(editor:GetEditorSource(entry.target) == entry.text, "No se sincronizo " .. entry[2])
 if entry.target:IsA("BaseScript") then entry.target.Enabled = true end
 print("[HR SINCRONIZADO] " .. entry[2])
end
local test = SSS:FindFirstChild("HR_PruebaBrazo")
if test and test:IsA("BaseScript") and test:GetAttribute("HRPrueba") == true then test.Enabled = false end
local old = SPS:FindFirstChild("HistoriasRobloxCliente")
if old and old:IsA("BaseScript") and old:GetAttribute("HRInstaller") == "HistoriasRoblox_E1_v1" then old.Enabled = false end
package.Parent = storage
folder:SetAttribute("HR_Version", "6.1-editor")
print("[HR SINCRONIZADO] Los tres scripts coinciden. Guarda, cierra sus pestanas y pulsa F5.")
